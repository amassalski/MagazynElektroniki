cmake_minimum_required(VERSION 3.16)
project(DomowyMagazyn VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# WYŁĄCZ ASSERTY - to rozwiąże problem linkowania
add_definitions(-DwxDEBUG_LEVEL=0 -DNDEBUG)

set(WXWIDGETS_ROOT "C:/wxWidgets-3.2.9")
set(WXWIDGETS_INCLUDE_DIR "${WXWIDGETS_ROOT}/include")
set(WXWIDGETS_LIB_DIR "${WXWIDGETS_ROOT}/build-mingw64-clean/lib")

include_directories(
    ${WXWIDGETS_INCLUDE_DIR}
    ${WXWIDGETS_ROOT}/build-mingw64-clean/lib/wx/include/msw-unicode-3.2
)

file(GLOB_RECURSE ALL_SRC "src/*.cpp")
add_executable(${PROJECT_NAME} WIN32 ${ALL_SRC})

target_link_directories(${PROJECT_NAME} PRIVATE ${WXWIDGETS_LIB_DIR})

# Użyj bibliotek które maszcd 
target_link_libraries(${PROJECT_NAME} PRIVATE
    wx_mswu_core-3.2
    wx_baseu-3.2
    kernel32 user32 gdi32 comdlg32 winspool winmm
    shell32 comctl32 ole32 oleaut32 uuid rpcrt4
    advapi32 wsock32 wininet
)

if(WIN32)
    add_definitions(-D_UNICODE -DUNICODE)
    set_target_properties(${PROJECT_NAME} PROPERTIES
        WIN32_EXECUTABLE TRUE
        LINK_FLAGS "-mwindows"
    )
endif()
